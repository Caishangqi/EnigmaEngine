# RenderCore.Tests: GoogleTest-based unit tests for RenderCore interfaces.
# Standalone CMake project, independent from BuildTool module system.
cmake_minimum_required(VERSION 3.28)
project(RenderCore.Tests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

enable_testing()

if(MSVC)
    add_compile_options(/std:c++latest /Zc:preprocessor)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-std=c++2c)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)

# ---------------------------------------------------------------------------
# Paths
# ---------------------------------------------------------------------------
set(ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Source/Runtime")
set(CORE_PUBLIC "${ENGINE_ROOT}/Core/Public")
set(CORE_PRIVATE "${ENGINE_ROOT}/Core/Private")
set(RENDERCORE_PUBLIC "${ENGINE_ROOT}/RenderCore/Public")
set(RENDERCORE_PRIVATE "${ENGINE_ROOT}/RenderCore/Private")
set(ENGINE_PUBLIC "${ENGINE_ROOT}/Engine/Public")
set(ENGINE_PRIVATE "${ENGINE_ROOT}/Engine/Private")
set(APPCORE_PUBLIC "${ENGINE_ROOT}/ApplicationCore/Public")
set(APPCORE_PRIVATE "${ENGINE_ROOT}/ApplicationCore/Private")
set(GOOGLETEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Source/ThirdParty/googletest")
set(NLOHMANN_JSON_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Source/ThirdParty/nlohmann_json/include")

# ---------------------------------------------------------------------------
# Shared source file lists
# ---------------------------------------------------------------------------
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/CoreSources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/ApplicationCoreSources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/RenderCoreSources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/EngineSources.cmake")

# ---------------------------------------------------------------------------
# Generated API headers
# ---------------------------------------------------------------------------
set(GENERATED_DIR "${CMAKE_BINARY_DIR}/Generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")

file(WRITE "${GENERATED_DIR}/CoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef CORE_STATIC_DEFINE\n"
"    #define CORE_API\n"
"#elif defined(CORE_EXPORTS)\n"
"    #define CORE_API DLLEXPORT\n"
"#else\n"
"    #define CORE_API DLLIMPORT\n"
"#endif\n"
)

file(WRITE "${GENERATED_DIR}/RenderCoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef RENDERCORE_STATIC_DEFINE\n"
"    #define RENDERCORE_API\n"
"#elif defined(RENDERCORE_EXPORTS)\n"
"    #define RENDERCORE_API DLLEXPORT\n"
"#else\n"
"    #define RENDERCORE_API DLLIMPORT\n"
"#endif\n"
)

file(WRITE "${GENERATED_DIR}/EngineAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef ENGINE_STATIC_DEFINE\n"
"    #define ENGINE_API\n"
"#elif defined(ENGINE_EXPORTS)\n"
"    #define ENGINE_API DLLEXPORT\n"
"#else\n"
"    #define ENGINE_API DLLIMPORT\n"
"#endif\n"
)

file(WRITE "${GENERATED_DIR}/ApplicationCoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef APPLICATIONCORE_STATIC_DEFINE\n"
"    #define APPLICATIONCORE_API\n"
"#elif defined(APPLICATIONCORE_EXPORTS)\n"
"    #define APPLICATIONCORE_API DLLEXPORT\n"
"#else\n"
"    #define APPLICATIONCORE_API DLLIMPORT\n"
"#endif\n"
)

# ---------------------------------------------------------------------------
# GoogleTest (gtest + gmock, static libraries)
# ---------------------------------------------------------------------------
add_subdirectory("${GOOGLETEST_DIR}" "${CMAKE_BINARY_DIR}/googletest")

# ---------------------------------------------------------------------------
# Core (SHARED)
# ---------------------------------------------------------------------------
set(CORE_SOURCES
    ${CORE_BASE_SOURCES}
    ${CORE_CONFIG_SOURCES}
    ${CORE_NAME_SOURCES}
)

add_library(Core SHARED ${CORE_SOURCES} ${CORE_MATH_SOURCES})
target_include_directories(Core PUBLIC "${CORE_PUBLIC}" "${GENERATED_DIR}")
target_compile_definitions(Core PRIVATE CORE_EXPORTS)

# ---------------------------------------------------------------------------
# ApplicationCore (SHARED)
# ---------------------------------------------------------------------------
add_library(ApplicationCore SHARED ${APPLICATIONCORE_SOURCES})
target_include_directories(ApplicationCore
    PUBLIC "${APPCORE_PUBLIC}" "${GENERATED_DIR}"
    PRIVATE "${APPCORE_PRIVATE}"
)
target_compile_definitions(ApplicationCore PRIVATE APPLICATIONCORE_EXPORTS)
target_link_libraries(ApplicationCore PUBLIC Core)

# ---------------------------------------------------------------------------
# RenderCore (STATIC for tests — avoids dllimport issues with mock classes)
# ---------------------------------------------------------------------------
add_library(RenderCore STATIC ${RENDERCORE_SOURCES})
target_include_directories(RenderCore
    PUBLIC "${RENDERCORE_PUBLIC}" "${GENERATED_DIR}"
)
target_compile_definitions(RenderCore PUBLIC RENDERCORE_STATIC_DEFINE)
target_link_libraries(RenderCore PUBLIC Core)

# ---------------------------------------------------------------------------
# Engine (SHARED)
# ---------------------------------------------------------------------------
add_library(Engine SHARED ${ENGINE_FULL_SOURCES})
target_include_directories(Engine
    PUBLIC "${ENGINE_PUBLIC}" "${GENERATED_DIR}"
    PRIVATE "${NLOHMANN_JSON_INCLUDE}"
)
target_compile_definitions(Engine PRIVATE ENGINE_EXPORTS)
target_link_libraries(Engine PUBLIC Core ApplicationCore RenderCore)

# ---------------------------------------------------------------------------
# Test sources
# ---------------------------------------------------------------------------
file(GLOB TEST_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")

# ---------------------------------------------------------------------------
# RenderCoreTests (executable, links all modules + GoogleTest)
# ---------------------------------------------------------------------------
add_executable(RenderCoreTests ${TEST_SOURCES})

target_include_directories(RenderCoreTests PRIVATE
    "${CORE_PUBLIC}"
    "${RENDERCORE_PUBLIC}"
    "${ENGINE_PUBLIC}"
    "${GENERATED_DIR}"
)

target_link_libraries(RenderCoreTests
    PRIVATE
        Core
        RenderCore
        Engine
        gtest
        gmock
        gtest_main
)

# ---------------------------------------------------------------------------
# CTest integration
# ---------------------------------------------------------------------------
include(GoogleTest)
gtest_discover_tests(RenderCoreTests)
