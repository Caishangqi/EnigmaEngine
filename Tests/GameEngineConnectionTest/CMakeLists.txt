# Task 4.2 Test: FGameEngine <-> FGameInstance frame loop connection
# Verifies: Tick drives frame loop, CreateGameInstance overridable,
# DeltaTime propagation, custom GameInstance subclass integration.
cmake_minimum_required(VERSION 3.28)
project(GameEngineConnectionTest LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
    add_compile_options(/std:c++latest /Zc:preprocessor)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-std=c++2c)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)

set(ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Source/Runtime")
set(CORE_PUBLIC "${ENGINE_ROOT}/Core/Public")
set(CORE_PRIVATE "${ENGINE_ROOT}/Core/Private")
set(ENGINE_PUBLIC "${ENGINE_ROOT}/Engine/Public")
set(ENGINE_PRIVATE "${ENGINE_ROOT}/Engine/Private")
set(APPCORE_PUBLIC "${ENGINE_ROOT}/ApplicationCore/Public")
set(APPCORE_PRIVATE "${ENGINE_ROOT}/ApplicationCore/Private")
set(RENDERCORE_PUBLIC "${ENGINE_ROOT}/RenderCore/Public")
set(NLOHMANN_JSON_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Source/ThirdParty/nlohmann_json/include")

# ---------------------------------------------------------------------------
# Shared source file lists
# ---------------------------------------------------------------------------
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/CoreSources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/ApplicationCoreSources.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/EngineSources.cmake")

# Generated API headers (normally produced by BuildTool)
set(GENERATED_DIR "${CMAKE_BINARY_DIR}/Generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}/Core/Public")
file(MAKE_DIRECTORY "${GENERATED_DIR}/Engine/Public")
file(MAKE_DIRECTORY "${GENERATED_DIR}/ApplicationCore/Public")
file(WRITE "${GENERATED_DIR}/Core/Public/CoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef CORE_STATIC_DEFINE\n"
"    #define CORE_API\n"
"#elif defined(CORE_EXPORTS)\n"
"    #define CORE_API DLLEXPORT\n"
"#else\n"
"    #define CORE_API DLLIMPORT\n"
"#endif\n"
)
file(WRITE "${GENERATED_DIR}/Engine/Public/EngineAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef ENGINE_STATIC_DEFINE\n"
"    #define ENGINE_API\n"
"#elif defined(ENGINE_EXPORTS)\n"
"    #define ENGINE_API DLLEXPORT\n"
"#else\n"
"    #define ENGINE_API DLLIMPORT\n"
"#endif\n"
)
file(WRITE "${GENERATED_DIR}/ApplicationCore/Public/ApplicationCoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef APPLICATIONCORE_STATIC_DEFINE\n"
"    #define APPLICATIONCORE_API\n"
"#elif defined(APPLICATIONCORE_EXPORTS)\n"
"    #define APPLICATIONCORE_API DLLEXPORT\n"
"#else\n"
"    #define APPLICATIONCORE_API DLLIMPORT\n"
"#endif\n"
)
file(MAKE_DIRECTORY "${GENERATED_DIR}/RenderCore/Public")
file(WRITE "${GENERATED_DIR}/RenderCore/Public/RenderCoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef RENDERCORE_STATIC_DEFINE\n"
"    #define RENDERCORE_API\n"
"#elif defined(RENDERCORE_EXPORTS)\n"
"    #define RENDERCORE_API DLLEXPORT\n"
"#else\n"
"    #define RENDERCORE_API DLLIMPORT\n"
"#endif\n"
)

# -- Core (SHARED) --
add_library(Core SHARED
    ${ENGINE_REQUIRED_CORE_SOURCES}
    ${CORE_MATH_SOURCES}
)
target_include_directories(Core PUBLIC "${CORE_PUBLIC}" "${GENERATED_DIR}/Core/Public")
target_compile_definitions(Core PRIVATE CORE_EXPORTS)

# -- ApplicationCore (SHARED) --
add_library(ApplicationCore SHARED ${APPLICATIONCORE_SOURCES})
target_include_directories(ApplicationCore PUBLIC "${APPCORE_PUBLIC}" "${GENERATED_DIR}/ApplicationCore/Public" PRIVATE "${APPCORE_PRIVATE}")
target_compile_definitions(ApplicationCore PRIVATE APPLICATIONCORE_EXPORTS)
target_link_libraries(ApplicationCore PUBLIC Core)

# -- Engine (SHARED) --
add_library(Engine SHARED ${ENGINE_FULL_SOURCES})
target_include_directories(Engine PUBLIC
    "${ENGINE_PUBLIC}"
    "${CORE_PUBLIC}"
    "${APPCORE_PUBLIC}"
    "${RENDERCORE_PUBLIC}"
    "${GENERATED_DIR}/Engine/Public"
    "${GENERATED_DIR}/ApplicationCore/Public"
    "${GENERATED_DIR}/RenderCore/Public"
    PRIVATE "${NLOHMANN_JSON_INCLUDE}"
)
target_link_libraries(Engine PUBLIC Core ApplicationCore)
target_compile_definitions(Engine PRIVATE ENGINE_EXPORTS)

# -- TestRunner --
add_executable(TestRunner Source/Main.cpp)
target_include_directories(TestRunner PRIVATE "${CORE_PUBLIC}" "${ENGINE_PUBLIC}" "${GENERATED_DIR}/Core/Public" "${GENERATED_DIR}/Engine/Public")
target_link_libraries(TestRunner PRIVATE Engine Core)

enable_testing()
add_test(NAME TestRunner COMMAND $<TARGET_FILE:TestRunner>)
