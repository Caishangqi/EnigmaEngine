# DLL Export Macro Test: verifies Platform.h, CoreAPI.h, EngineAPI.h, LaunchAPI.h
cmake_minimum_required(VERSION 3.28)
project(DllExportMacroTest LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
    add_compile_options(/std:c++latest)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-std=c++2c)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)

# Engine module public header paths
set(CORE_PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Source/Runtime/Core/Public")

# Generated API headers (normally produced by BuildTool)
set(GENERATED_DIR "${CMAKE_BINARY_DIR}/Generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}/Core/Public")
file(MAKE_DIRECTORY "${GENERATED_DIR}/Engine/Public")
file(MAKE_DIRECTORY "${GENERATED_DIR}/Launch/Public")
file(WRITE "${GENERATED_DIR}/Core/Public/CoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef CORE_STATIC_DEFINE\n"
"    #define CORE_API\n"
"#elif defined(CORE_EXPORTS)\n"
"    #define CORE_API DLLEXPORT\n"
"#else\n"
"    #define CORE_API DLLIMPORT\n"
"#endif\n"
)
file(WRITE "${GENERATED_DIR}/Engine/Public/EngineAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef ENGINE_STATIC_DEFINE\n"
"    #define ENGINE_API\n"
"#elif defined(ENGINE_EXPORTS)\n"
"    #define ENGINE_API DLLEXPORT\n"
"#else\n"
"    #define ENGINE_API DLLIMPORT\n"
"#endif\n"
)
file(WRITE "${GENERATED_DIR}/Launch/Public/LaunchAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef LAUNCH_STATIC_DEFINE\n"
"    #define LAUNCH_API\n"
"#elif defined(LAUNCH_EXPORTS)\n"
"    #define LAUNCH_API DLLEXPORT\n"
"#else\n"
"    #define LAUNCH_API DLLIMPORT\n"
"#endif\n"
)

# Test executable -- compiled as a consumer (no _EXPORTS defined)
add_executable(DllExportMacroTest Source/Main.cpp)
target_include_directories(DllExportMacroTest PRIVATE "${CORE_PUBLIC}" "${GENERATED_DIR}/Core/Public")

enable_testing()
add_test(NAME DllExportMacroTest COMMAND $<TARGET_FILE:DllExportMacroTest>)
