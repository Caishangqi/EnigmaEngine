# DirectoryWatcher.Tests: GoogleTest-based unit tests for DirectoryWatcher module.
# Standalone CMake project, independent from BuildTool module system.
cmake_minimum_required(VERSION 3.28)
project(DirectoryWatcher.Tests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

enable_testing()

if(MSVC)
    add_compile_options(/std:c++latest /Zc:preprocessor)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-std=c++2c)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Bin)

# ---------------------------------------------------------------------------
# Paths
# ---------------------------------------------------------------------------
set(ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Source")
set(CORE_PUBLIC "${ENGINE_ROOT}/Runtime/Core/Public")
set(CORE_PRIVATE "${ENGINE_ROOT}/Runtime/Core/Private")
set(DW_ROOT "${ENGINE_ROOT}/Developer/DirectoryWatcher")
set(DW_PUBLIC "${DW_ROOT}/Public")
set(DW_PRIVATE "${DW_ROOT}/Private")
set(GOOGLETEST_DIR "${ENGINE_ROOT}/ThirdParty/googletest")

# ---------------------------------------------------------------------------
# Generated headers
# ---------------------------------------------------------------------------
set(GENERATED_DIR "${CMAKE_BINARY_DIR}/Generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")

file(WRITE "${GENERATED_DIR}/CoreAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef CORE_STATIC_DEFINE\n"
"    #define CORE_API\n"
"#elif defined(CORE_EXPORTS)\n"
"    #define CORE_API DLLEXPORT\n"
"#else\n"
"    #define CORE_API DLLIMPORT\n"
"#endif\n"
)

file(WRITE "${GENERATED_DIR}/DirectoryWatcherAPI.generated.h"
"// Auto-generated by CMake for standalone test builds.\n"
"#pragma once\n"
"\n"
"#include \"HAL/Platform.h\"\n"
"\n"
"#ifdef DIRECTORYWATCHER_STATIC_DEFINE\n"
"    #define DIRECTORYWATCHER_API\n"
"#elif defined(DIRECTORYWATCHER_EXPORTS)\n"
"    #define DIRECTORYWATCHER_API DLLEXPORT\n"
"#else\n"
"    #define DIRECTORYWATCHER_API DLLIMPORT\n"
"#endif\n"
)

# ---------------------------------------------------------------------------
# GoogleTest
# ---------------------------------------------------------------------------
add_subdirectory("${GOOGLETEST_DIR}" "${CMAKE_BINARY_DIR}/googletest")

# ---------------------------------------------------------------------------
# Core (SHARED)
# ---------------------------------------------------------------------------
set(CORE_SOURCES
    "${CORE_PRIVATE}/Modules/ModuleInitializerEntry.cpp"
    "${CORE_PRIVATE}/Modules/ModuleManager.cpp"
    "${CORE_PRIVATE}/Logging/LogSystem.cpp"
    "${CORE_PRIVATE}/Misc/AssertionMacros.cpp"
    "${CORE_PRIVATE}/CoreModule.cpp"
    "${CORE_PRIVATE}/Delegates/DelegateHandle.cpp"
    "${CORE_PRIVATE}/Containers/Ticker.cpp"
)

file(GLOB CORE_MATH_SOURCES "${CORE_PRIVATE}/Math/*.cpp")

add_library(Core SHARED ${CORE_SOURCES} ${CORE_MATH_SOURCES})
target_include_directories(Core PUBLIC "${CORE_PUBLIC}" "${GENERATED_DIR}")
target_compile_definitions(Core PRIVATE CORE_EXPORTS)

# ---------------------------------------------------------------------------
# DirectoryWatcher (STATIC for test linking)
# ---------------------------------------------------------------------------
set(DW_SOURCES
    "${DW_PRIVATE}/DirectoryWatcherProxy.cpp"
    "${DW_PRIVATE}/DirectoryWatcherModule.cpp"
    "${DW_PRIVATE}/Windows/DirectoryWatchRequestWindows.cpp"
    "${DW_PRIVATE}/Windows/DirectoryWatcherWindows.cpp"
)

add_library(DirectoryWatcher STATIC ${DW_SOURCES})
target_include_directories(DirectoryWatcher
    PUBLIC "${DW_PUBLIC}"
    PRIVATE "${DW_PRIVATE}" "${CORE_PUBLIC}" "${GENERATED_DIR}"
)
target_compile_definitions(DirectoryWatcher PUBLIC DIRECTORYWATCHER_STATIC_DEFINE)
target_link_libraries(DirectoryWatcher PUBLIC Core)

# ---------------------------------------------------------------------------
# Test sources
# ---------------------------------------------------------------------------
file(GLOB TEST_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")

# ---------------------------------------------------------------------------
# DirectoryWatcherTests executable
# ---------------------------------------------------------------------------
add_executable(DirectoryWatcherTests ${TEST_SOURCES})

target_include_directories(DirectoryWatcherTests PRIVATE
    "${DW_PUBLIC}" "${DW_PRIVATE}" "${CORE_PUBLIC}"
)

target_link_libraries(DirectoryWatcherTests
    PRIVATE
        DirectoryWatcher
        Core
        gtest
        gmock
        gtest_main
)

# ---------------------------------------------------------------------------
# CTest integration
# ---------------------------------------------------------------------------
include(GoogleTest)
gtest_discover_tests(DirectoryWatcherTests)
